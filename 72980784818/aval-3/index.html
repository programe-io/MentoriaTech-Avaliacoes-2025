document.getElementById('postButton').addEventListener('click', function () { const content = document.getElementById('postContent').value.trim(); if (content === '') { alert('Por favor, escreva algo antes de postar!'); return; } const postDiv = document.createElement('div'); postDiv.classList.add('post'); // Criar o texto do post com a palavra "post" destacada const postText = document.createElement('p'); // Exemplo simples: substituir a palavra "post" por um span colorido const highlightedContent = content.replace(/post/gi, 'post'); postText.innerHTML = highlightedContent; postDiv.appendChild(postText); const postsContainer = document.getElementById('postsContainer'); postsContainer.insertBefore(postDiv, postsContainer.firstChild); document.getElementById('postContent').value = ''; });
