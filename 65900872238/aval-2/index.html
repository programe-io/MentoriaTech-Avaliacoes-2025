document.addEventListener('DOMContentLoaded',()=>{ const themeBtn = document.getElementById('theme-toggle'); const body = document.body; // carregamento do tema salvo if(localStorage.getItem('tema') === 'claro'){ body.classList.add('tema-claro'); themeBtn.textContent = '☀️'; themeBtn.setAttribute('aria-pressed','true'); } themeBtn.addEventListener('click',()=>{ const ativo = body.classList.toggle('tema-claro'); themeBtn.setAttribute('aria-pressed', String(ativo)); themeBtn.textContent = ativo ? '☀️' : '🌙'; localStorage.setItem('tema', ativo ? 'claro' : 'escuro'); }); // Player: atualizar progress const audio = document.getElementById('audio'); const progress = document.getElementById('progress'); const volume = document.getElementById('volume'); const favBtn = document.getElementById('fav-btn'); if(audio){ audio.addEventListener('timeupdate',()=>{ const pct = (audio.currentTime / audio.duration) * 100 || 0; progress.value = pct; }); } volume?.addEventListener('input',(e)=>{ if(audio) audio.volume = parseFloat(e.target.value); }); // favoritos const FAVORITES_KEY = 'kpfav'; favBtn?.addEventListener('click',()=>{ const favs = JSON.parse(localStorage.getItem(FAVORITES_KEY) || '[]'); const track = audio?.currentSrc || 'track-1'; if(favs.includes(track)){ const idx = favs.indexOf(track); if(idx>-1) favs.splice(idx,1); favBtn.textContent = '♡ Favoritar'; favBtn.setAttribute('aria-pressed','false'); showModal('Removido dos favoritos'); } else { favs.push(track); favBtn.textContent = '♥ Favoritado'; favBtn.setAttribute('aria-pressed','true'); showModal('Adicionado aos favoritos'); } localStorage.setItem(FAVORITES_KEY, JSON.stringify(favs)); }); // Form submit (validação simples) const form = document.getElementById('contact-form'); form?.addEventListener('submit',(e)=>{ e.preventDefault(); const data = new FormData(form); const nome = data.get('name')?.toString().trim(); const email = data.get('email')?.toString().trim(); if(!nome || !email){ showModal('Por favor, preencha nome e email.'); return; } showModal('Obrigado! Mensagem enviada (demo).'); form.reset(); }); // Modal utilitário const modal = document.getElementById('modal'); const modalText = document.getElementById('modal-text'); const modalClose = document.getElementById('modal-close'); window.showModal = (msg)=>{ if(!modal) return; modalText.textContent = msg; modal.setAttribute('aria-hidden','false'); // foco acessível modalClose.focus(); } modalClose?.addEventListener('click',()=>{ modal.setAttribute('aria-hidden','true'); }); // fechar modal com ESC document.addEventListener('keydown',(e)=>{ if(e.key === 'Escape') modal?.setAttribute('aria-hidden','true'); }); // Keyboard shortcuts: / abre busca (demo) document.addEventListener('keydown',(e)=>{ if(e.key === '/'){ e.preventDefault(); showModal('Atalho: você pressionou / — busque no site (demo)'); } }); }); ``` --- Se quiser, eu posso: * Gerar um arquivo ZIP com estes arquivos. * Acrescentar mais páginas (ex: `sobre.html`, `discografia.html`) ou designs diferentes. * Traduzir para inglês ou outro idioma.
